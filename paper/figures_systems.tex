% loop above the span [#1,#2] in row #3 with label #4
\newcommand{\iterloop}[4]{%
  \draw[loop] ({#2-3},{#3+\hRow/2}) -- ({#2-3},{#3+\hRow/2+2.8}) -- ({#1+3},{#3+\hRow/2+2.8}) -- ({#1+3},{#3+\hRow/2+0.3});
  \node[lbl, anchor=south] at ({(#1+#2)/2},{#3+\hRow/2+3.1}) {#4};}

% ============================ row 1: FEDOT.LLM ==========================
\node[lane, text=cFedot] at (\xLane,\yA) {\textbf{FEDOT.LLM}\\[-1pt]{\color{muted}fixed-stage workflow}};
\node[io] (t1) at (\xTask,\yA) {task\\[-2pt]{\color{muted}tabular}};
\node[stage, anchor=west] (a1) at (\xA,\yA) {understand\\the data};
\node[stage, anchor=west] (b1) at (\xB,\yA) {configure\\FEDOT};
\node[stage, anchor=west] (c1) at (\xC,\yA) {fill the\\code template};
\node[stage, anchor=west] (d1) at (\xD,\yA) {run and\\repair};
\node[io] (e1) at (\xOut,\yA) {submission};
\foreach \i/\j in {t1/a1,a1/b1,b1/c1,c1/d1,d1/e1} \draw[arr] (\i) -- (\j);
\iterloop{\xD}{\xD+\wS}{\yA}{until the code passes or the retry cap}
\node[tag, anchor=north] at ([yshift=-1.2mm]b1.south) {library instruction: built in};

% ============================ row 2: AutoDS-Tools ========================
\node[lane, text=cAuto] at (\xLane,\yB) {\textbf{AutoDS-Tools}\\[-1pt]{\color{muted}six-agent system}};
\node[io] (t2) at (\xTask,\yB) {task\\[-2pt]{\color{muted}any}};
\node[stage, anchor=west] (a2) at (\xA,\yB) {Analyst,\\Researcher};
\node[free, draw=cAuto, anchor=west] (b2) at (\xB,\yB) {Planner};
\node[free, draw=cAuto, anchor=west] (c2) at (\xC,\yB) {Coder\\{\color{muted}any library}};
\node[stage, anchor=west] (d2) at (\xD,\yB) {Debugger,\\Presenter};
\node[io] (e2) at (\xOut,\yB) {submission};
\foreach \i/\j in {t2/a2,a2/b2,b2/c2,c2/d2,d2/e2} \draw[arr] (\i) -- (\j);
\iterloop{\xC}{\xD+\wS}{\yB}{per-agent retries}
\node[tag, anchor=north] (k2) at ({(\xTask+\xA+8)/2},{\yB-7}) {instruction file: on / off};
\draw[arr, draw=muted] (k2.north) -- ++(0,2.2) -| (t2.south);

% ============================ row 3: Terminus-2 ==========================
\node[lane, text=cTerm] at (\xLane,\yC) {\textbf{Terminus-2}\\[-1pt]{\color{muted}single agent, shell}};
\node[io] (t3) at (\xTask,\yC) {task\\[-2pt]{\color{muted}any}};
\node[free, draw=cTerm, anchor=west, minimum width={(\xD+\wS-\xA)*1mm}, text width={(\xD+\wS-\xA-3)*1mm}] (a3) at (\xA,\yC)
